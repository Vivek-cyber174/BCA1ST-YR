    <!-- 2. Colour + size -->
    <h2 style="color: green; font-size: 28px;">Inline CSS Examples</h2>

    <!-- 3. Center alignment -->
    <p style="text-align: center;">This paragraph is in the center.</p>

    <!-- 4. Background colour -->
    <p style="background-color: yellow;">This paragraph has a yellow background.</p>

    <!-- 5. Font family and style -->
    <p style="font-family: 'Times New Roman'; font-style: italic;">This text is italic in Times New Roman.</p>

    <!-- 6. Bold and uppercase -->
    <p style="font-weight: bold; text-transform: uppercase;">this text becomes bold and capital</p>

    <!-- 7. Underline and line-through -->
    <p style="text-decoration: underline;">This text is underlined.</p>
    <p style="text-decoration: line-through; color: red;">Old Price: Rs. 999</p>

    <!-- 8. Border and padding -->
    <p style="border: 2px solid black; padding: 10px;">This paragraph has a border and inner space.</p>

    <!-- 9. Box with width and margin -->
    <div style="width: 300px; background-color: lightpink; border: 2px dashed purple; padding: 15px; margin: 20px;">
        This is a styled box.
    </div>

    <!-- 10. Styling one word with span -->
    <p>My result is <span style="color: red; font-weight: bold;">PASS</span>.</p>

    <!-- 11. Link without underline -->
    <a href="https://www.google.com" style="color: darkorange; text-decoration: none;">Go to Google</a>

    <!-- 12. Round image -->
    <br><br>
    <img src="photo.jpg" alt="Photo" style="width: 150px; height: 150px; border-radius: 50%; border: 3px solid navy;">

    <!-- 13. Button -->
    <br><br>
    <button style="background-color: green; color: white; padding: 10px 20px; border: none; border-radius: 5px;">Submit</button>
